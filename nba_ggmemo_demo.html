Top 10 Scorers

Horizontal bar chart with annotate_callout() highlighting Luka Dončić as scoring champ and Nikola Jokić’s triple-double averages.

Code 
top10 <- nba[1:10, ]
top10$PLAYER <- factor(top10$PLAYER, levels = rev(top10$PLAYER))

small_arrow <- grid::arrow(length = grid::unit(0.12, "inches"))

ggplot(top10, aes(y = PLAYER, x = PTS)) +
  geom_col(fill = "#1D428A", width = 0.7) +
  annotate_callout(
    top10,
    where = RANK == 1,
    label = "Scoring champ",
    position = "top-right",
    nudge = c(2, 0.4),
    arrow = small_arrow
  ) +
  annotate_callout(
    top10,
    where = PLAYER == "Nikola Jokić",
    label = "Triple-double machine\n(12.9 REB, 10.7 AST)",
    position = "top-right",
    nudge = c(3, 0.4),
    size = 3.5,
    arrow = small_arrow
  ) +
  scale_x_continuous(expand = expansion(mult = c(0, 0.15))) +
  coord_cartesian(clip = "off") +
  labs(y = NULL, x = "PPG") +
  theme_minimal(base_size = 14) +
  theme(plot.margin = margin(10, 80, 10, 10))

Field Goal Efficiency

annotate_change(format = "points") shows the 11.1 percentage-point gap between Zion Williamson (60.0%) and Jalen Johnson (48.9%).

Code 
fg_top <- nba[order(-nba$FG_PCT), ][1:10, ]
fg_top$FG_PCT_100 <- fg_top$FG_PCT * 100
fg_top$PLAYER <- factor(fg_top$PLAYER, levels = fg_top$PLAYER)

fg_simple <- fg_top[, c("PLAYER", "FG_PCT_100")]

ggplot(fg_top, aes(x = PLAYER, y = FG_PCT_100)) +
  geom_col(fill = "#C8102E", width = 0.6) +
  annotate_change(
    fg_simple,
    from = PLAYER == levels(fg_simple$PLAYER)[1],
    to = PLAYER == levels(fg_simple$PLAYER)[10],
    value = FG_PCT_100,
    format = "points"
  ) +
  labs(x = NULL, y = "FG%") +
  theme_minimal(base_size = 14) +
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

Efficiency Rating

Lollipop (Cleveland dot) chart showing annotate_callout() works beyond bar charts. SGA is called out as 2025-26 MVP, while Luka’s callout uses colour = "#C8102E" to highlight the efficiency gap despite leading in scoring.

Code 
nba_eff <- nba[order(-nba$EFF), ]
nba_eff$PLAYER <- factor(nba_eff$PLAYER, levels = rev(nba_eff$PLAYER))

ggplot(nba_eff[1:15, ], aes(x = EFF, y = PLAYER)) +
  geom_segment(aes(x = 0, xend = EFF, yend = PLAYER),
               color = "grey70", linewidth = 0.4) +
  geom_point(size = 4, color = "#552583") +
  annotate_callout(
    nba_eff[1:15, ],
    where = PLAYER == "Shai Gilgeous-Alexander",
    label = "2025-26 MVP",
    position = "top-right",
    nudge = c(3, 0.5),
    size = 3.5
  ) +
  annotate_callout(
    nba_eff[1:15, ],
    where = PLAYER == "Luka Dončić",
    label = "Scoring champ but\nlower efficiency (33.5 PPG)",
    position = "bottom-left",
    nudge = c(5, 1.5),
    size = 3.5,
    colour = "#C8102E"
  ) +
  labs(
    title = "NBA Efficiency Rating — Top 15 Players",
    subtitle = "Points + rebounds + assists + steals + blocks - missed shots - turnovers",

Rebounding: Wemby vs Flagg

Head-to-head comparison using annotate_change(format = "percent") for the percentage difference, combined with annotate_callout() on both bars.

Code 
duo <- nba[nba$PLAYER %in% c("Victor Wembanyama", "Cooper Flagg"), ]
duo$PLAYER <- factor(duo$PLAYER, levels = c("Cooper Flagg", "Victor Wembanyama"))
duo_simple <- duo[, c("PLAYER", "REB")]

ggplot(duo, aes(x = PLAYER, y = REB)) +
  geom_col(fill = c("#00538C", "#C4CED4"), width = 0.5) +
  annotate_change(
    duo_simple,
    from = PLAYER == "Cooper Flagg",
    to = PLAYER == "Victor Wembanyama",
    value = REB,
    format = "percent"
  ) +
  annotate_callout(
    duo,
    where = PLAYER == "Victor Wembanyama",
    label = "Elite rebounder at 7'4\"",
    position = "top-left",
    nudge = c(0.15, 0.6)
  ) +
  annotate_callout(
    duo,
    where = PLAYER == "Cooper Flagg",
    label = "Rookie: 6.7 RPG",
    position = "top-left",
    nudge = c(0.15, 0.6),
    size = 3.5
  ) +
  labs(
    title = "Rebounding: Wemby vs Flagg",
    subtitle = "Per-game rebounds, 2025-26 season",
    x = NULL, y = "Rebounds per Game"
  ) +
  theme_minimal(base_size = 14)
